⚛️ React Integration Example 

import { io } from 'socket.io-client';
import { useEffect, useState } from 'react';

function useNotifications() {
  const [socket, setSocket] = useState(null);
  const [notifications, setNotifications] = useState([]);
  const [unreadCount, setUnreadCount] = useState(0);

  useEffect(() => {
    const token = localStorage.getItem('token');
    if (!token) return;

    // Connect to WebSocket with namespace '/notifications'
    const newSocket = io('http://localhost:5000/notifications', {
      auth: { token: `Bearer ${token}` }
    });

    // Listen for new notifications
    newSocket.on('notification:new', (notification) => {
      setNotifications(prev => [notification, ...prev]);
      // Show toast/alert
    });

    // Listen for unread count updates
    newSocket.on('notification:unread_count', ({ count }) => {
      setUnreadCount(count);
    });

    setSocket(newSocket);
    
    return () => newSocket.disconnect();
  }, []);

  const loadNotifications = (page = 1, limit = 20) => {
    socket?.emit('notification:load', { page, limit });
  };

  const markAsRead = (notificationId) => {
    socket?.emit('notification:mark_read', { notificationId });
  };

  return { notifications, unreadCount, loadNotifications, markAsRead };
}

🟢 Vue Integration Example 

import { io } from 'socket.io-client';
import { ref, onMounted, onUnmounted } from 'vue';

export function useNotifications() {
  const socket = ref(null);
  const notifications = ref([]);
  const unreadCount = ref(0);

  const connect = () => {
    const token = localStorage.getItem('token');
    
    socket.value = io('http://localhost:5000/notifications', {
      auth: { token: `Bearer ${token}` }
    });

    socket.value.on('notification:new', (notification) => {
      notifications.value.unshift(notification);
    });

    socket.value.on('notification:unread_count', ({ count }) => {
      unreadCount.value = count;
    });
  };

  onMounted(connect);
  onUnmounted(() => socket.value?.disconnect());

  return { notifications, unreadCount };
}

🅰️ Angular Integration Example 

import { Injectable } from '@angular/core';
import { io, Socket } from 'socket.io-client';
import { BehaviorSubject } from 'rxjs';

@Injectable({ providedIn: 'root' })
export class NotificationService {
  private socket: Socket;
  public notifications$ = new BehaviorSubject([]);
  public unreadCount$ = new BehaviorSubject(0);

  connect(token: string) {
    this.socket = io('http://localhost:5000/notifications', {
      auth: { token: `Bearer ${token}` }
    });

    this.socket.on('notification:new', (notification) => {
      const current = this.notifications$.value;
      this.notifications$.next([notification, ...current]);
    });

    this.socket.on('notification:unread_count', ({ count }) => {
      this.unreadCount$.next(count);
    });
  }
}

📡 WebSocket Events Reference

Client → Server Events (Emit)

Event Payload Description
notification:load { page?: number, limit?: number } Load notifications with pagination
notification:mark_read { notificationId: string } Mark single notification as read
notification:mark_all_read - Mark all notifications as read
notification:delete { notificationId: string } Delete a notification

Server → Client Events (Listen)

Event Payload Description
notification:new Notification object New notification received
notification:list { data: [], metadata: {} } Notifications list response
notification:unread_count { count: number } Unread count update
notification:success { userId: string } Connection successful
notification:error { message: string } Error occurred

🚀 Quick Start Steps

  1. Install Socket.IO Client: 
    npm install socket.io-client
  2. Import and Connect: Use the namespace /notifications
  3. Authenticate: Pass JWT token in auth config with Bearer prefix
  4. Listen for Events: Subscribe to notification:new and notification:unread_count
  5. Load Notifications: Emit notification:load on component mount
  6. Handle Actions: Emit events to mark as read or delete notifications

⚠️ Important Notes

  • Namespace: Always connect to /notifications namespace, not root
  • Token Format: Must include "Bearer " prefix: Bearer YOUR_TOKEN
  • Connection: One connection per user session
  • Disconnect: Clean up socket connection on component unmount
  • Error Handling: Listen to notification:error for auth/connection errors
  • Real-time Updates: Unread count updates automatically when marking as read
